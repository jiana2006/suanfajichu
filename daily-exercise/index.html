    #include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int n, k;
    if (!(cin >> n >> k)) return 0;

    // 使用 unordered_map 记录每个案件的最早位置、最晚位置、总页数
    unordered_map<int, int> min_pos, max_pos;
    unordered_map<int, ll> sum_w;

    for (int i = 1; i <= n; i++) {
        int c; ll w;
        cin >> c >> w;
        if (min_pos.find(c) == min_pos.end()) {
            min_pos[c] = i;
        }
        max_pos[c] = i;
        sum_w[c] += w;
    }

    // 提取所有案件，按最早出现位置排序
    vector<tuple<int, int, ll>> cases;
    for (auto& [c, w] : sum_w) {
        cases.push_back({min_pos[c], max_pos[c], w});
    }
    sort(cases.begin(), cases.end());

    // 区间合并，得到不可拆分的“大块”
    vector<ll> blocks;
    int curL = -1, curR = -1;
    ll curSum = 0;

    for (auto [L, R, W] : cases) {
        if (curL == -1) {
            curL = L; curR = R; curSum = W;
        } else if (L <= curR) {
            // 有重叠，合并
            curR = max(curR, R);
            curSum += W;
        } else {
            // 无重叠，结算上一个块
            blocks.push_back(curSum);
            curL = L; curR = R; curSum = W;
        }
    }
    if (curL != -1) blocks.push_back(curSum);

    int m = blocks.size();
    // 如果合并后的块数超过 k，无解
    if (m > k) {
        cout << -1 << "\n";
        return 0;
    }

    // 二分答案
    ll low = 0, high = 0;
    for (ll w : blocks) {
        low = max(low, w); // 下界是最大的单块页数
        high += w;         // 上界是总页数
    }

    ll ans = high;
    while (low <= high) {
        ll mid = low + (high - low) / 2;

        // check(mid): 能否在每册不超过 mid 的情况下，划分为 <= k 册
        int cnt = 0;     // 需要的册数
        ll cur = 0;      // 当前册的页数
        for (ll w : blocks) {
            if (cur + w > mid) {
                cnt++;
                cur = w;
            } else {
                cur += w;
            }
        }
        if (cur > 0) cnt++; // 最后一册

        if (cnt <= k) {
            ans = mid;
            high = mid - 1; // 尝试更小的最大值
        } else {
            low = mid + 1;  // 限制太严，需要放宽
        }
    }

    cout << ans << "\n";
    return 0;
}

    一次移动后能否使 A 与 B 一致

    一次移动后能否使 A 与 B 一致

    问题描述

    给定由正整数组成、长度为 N 的序列 A=(A1,A2,…,AN) 和 B=(B1,B2,…,BN)。

    对序列 A 恰好进行 1 次以下操作。

    从 A 中选出一个元素取出，并将其插入剩余序列的任意位置。 更严格地说，依次进行以下 2 个操作。

    1. 选择一个满足 1 ≤ i ≤ N 的整数 i， 令长度为 N−1 的序列 X=(X1,X2,…,XN−1) 为 (A1,A2,…,Ai−1,Ai+1,Ai+2,…,AN)， 并令整数 Y=Ai。
    2. 选择一个满足 1 ≤ j ≤ N 的整数 j， 将长度为 N 的序列 (X1,X2,…,Xj−1,Y,Xj,Xj+1,…,XN−1) 作为新的 A。

    注意，插入元素时，也可以插入到序列的左端或右端。

    请判断操作之后能否使 A 与 B 一致。

    约束

    • 1 ≤ N ≤ 2×105
    • 1 ≤ Ai ≤ N（1 ≤ i ≤ N）
    • 1 ≤ Bi ≤ N（1 ≤ i ≤ N）
    • 输入均为整数

    输入

    输入以如下格式从标准输入给出。

    N
A1 A2 … AN
B1 B2 … BN

    输出

    如果恰好进行 1 次操作后能使 A 与 B 一致，则输出一行 Yes； 否则输出一行 No。

    输入样例 1

    6
1 4 1 4 2 1
1 4 1 1 4 2

    输出样例 1

    Yes

    例如，取出 A6，并将其插入剩余序列从头数第 3 项和第 4 项之间， 则操作后的 A 可以变为 (1,4,1,1,4,2)。

    因此输出 Yes。

    输入样例 2

    2
1 1
1 2

    输出样例 2

    No

    无论对 A 进行怎样的操作，都无法使其与 B 一致。

    输入样例 3

    3
1 2 3
1 2 3

    输出样例 3

    Yes

    将取出的元素插入到原本所在的位置也可以。

    输入样例 4

25 11 21 15 30 20 17 19 17 21 10 13 22 10 22 14 30 25 17 25 11 19 27 27 15 18 16 16 20 16

    Yes

    #include<bits/stdc++.h>
using namespace std;
#define endl '\n'

int main(){
    ios::sync_with_stdio(0);
    cin.tie(0);
    int n;
    cin >> n;
    vector<int> v1(n), v2(n);
    for (int i = 0; i < n; i++) cin >> v1[i];
    for (int i = 0; i < n; i++) cin >> v2[i];

    if (v1 == v2){
        cout << "Yes" << endl;
        return 0;
    }
    int l = 0;
    while (l < n && v1[l] == v2[l]) l++;
    int r = n - 1;
    while (r >= 0 && v1[r] == v2[r]) r--;
    bool f = (v1[l] == v2[r]);
    for (int i = l; i < r && f; i++){
        if (v1[i + 1] != v2[i]) f = false;
    }
    bool f1 = (v1[r] == v2[l]);
    for (int i = l; i < r && f1; i++){
        if (v1[i] != v2[i + 1]) f1 = false;
    }

    if (f || f1){
        cout << "Yes" << endl;
    } else {
        cout << "No" << endl;
    }
    return 0;
}

    操作后可能得到的序列种数

    操作后可能得到的序列种数

    问题描述

    给定一个由正整数组成、长度为 N 的序列 A=(A1,A2,…,AN)。

    对序列 A 恰好进行 1 次以下操作。

    从 A 中选出一个元素取出，并将它插入到剩余序列中原来的位置或比原来更靠前的位置。 更严格地说，依次进行以下 2 个操作。

    1. 选择一个满足 1 ≤ i ≤ N 的整数 i， 令长度为 N−1 的序列 X=(X1,X2,…,XN−1) 为 (A1,A2,…,Ai−1,Ai+1,Ai+2,…,AN)， 并令整数 Y=Ai。
    2. 选择一个满足 1 ≤ j ≤ i 的整数 j， 将长度为 N 的序列 (X1,X2,…,Xj−1,Y,Xj,Xj+1,…,XN−1) 作为新的 A。

    注意，插入元素时，有时可以插入到序列的左端或右端。

    请回答：作为操作结果，可能得到多少种不同的序列？

    即使选择的是不同的 (i,j)，如果得到的序列相同，也只算作同一种。

    约束

    • 输入均为整数
    • 1 ≤ N ≤ 2×105
    • 1 ≤ Ai ≤ N（1 ≤ i ≤ N）

    输入

    输入以如下格式从标准输入给出。

    N
A1 A2 … AN

    输出

    输出答案。

    输入样例 1

    6
1 4 1 4 2 1

    输出样例 1

    12

    例如，取 i=6,j=4 进行操作，可以得到 (1,4,1,1,4,2)。

    除此之外，还可以得到以下 11 种序列。 

    (1,1,4,1,4,2)
(1,1,4,4,2,1)
(1,2,4,1,4,1)
(1,4,1,2,4,1)
(1,4,1,4,1,2)
(1,4,1,4,2,1)
(1,4,2,1,4,1)
(1,4,4,1,2,1)
(2,1,4,1,4,1)
(4,1,1,4,2,1)
(4,1,4,1,2,1)

    因此输出 12。

    输入样例 2

    5
1 1 1 1 1

    输出样例 2

    1

    注意，即使由不同的 (i,j) 得到，只要序列相同，就只计作 1 种。

    输入样例 3

    30
24 22 18 20 14 22 14 14 26 14 26 26 20 22 24 18 22 22 14 24 18 14 14 24 22 18 18 26 14 20

    输出样例 3

    303

    580C - 凯法和公园

    580C - 凯法和公园（Kefa and Park）

    时间限制：2 秒

    内存限制：256 MB

    题目描述

    凯法（Kefa）决定用自己第一份丰厚的薪水去餐厅庆祝。

    他住在一个不寻常的公园旁边。这个公园是一棵有 n 个顶点的有根树，根是顶点 1。顶点 1 也是凯法的家。

    不幸的是，公园里还有猫。凯法已经知道哪些顶点上有猫。

    公园的叶子顶点上开有餐厅。凯法想选一个餐厅去吃饭，但他非常怕猫。因此，如果从餐厅到他家的路径上，包含超过 m 个连续有猫的顶点，他就不会去这个餐厅。

    你的任务是帮助凯法计算：他一共可以去多少个餐厅。

    输入格式

    第一行包含两个整数 n 和 m（2 ≤ n ≤ 105，1 ≤ m ≤ n），分别表示树的顶点数，以及凯法最多能接受的连续有猫顶点数。

    第二行包含 n 个整数 a1, a2, ..., an。若 ai = 0，表示顶点 i 没有猫；若 ai = 1，表示顶点 i 有猫。

    接下来 n − 1 行，每行包含两个整数 xi 和 yi（1 ≤ xi, yi ≤ n，xi ≠ yi），表示树中顶点 xi 和 yi 之间有一条边。

    保证给出的边构成一棵树。

    输出格式

    输出一个整数，表示从凯法家出发，路径上连续有猫的顶点数不超过 m 的叶子餐厅数量。

    样例 1

    输入

    4 1
1 1 0 0
1 2
1 3
1 4

    输出

    2

    样例 2

    输入

    7 1
1 0 1 1 0 0 0
1 2
1 3
2 4
2 5
3 6
3 7

    输出

    2

    说明

    树是含有 n 个顶点和 n − 1 条边的连通图。有根树是指定一个特殊顶点作为根的树。在有根树中，对于一条边连接的两个顶点，离根更近的那个叫父节点，另一个叫子节点。没有子节点的顶点称为叶子。

    第一个样例中，有猫的顶点被标记出来。餐厅在顶点 2、3、4。凯法不能去顶点 2 的餐厅。

    第二个样例中，餐厅在顶点 4、5、6、7。凯法不能去餐厅 6 和 7。

    atcode580C.cpp
    #include<bits/stdc++.h>
using namespace std;

#define endl "\n"

const int N = 1e5 + 10;
const int M = 2 * N;

int h[N], nxt[M], t[M], cnt;
int a[N];

void add (int a, int b){
    t[cnt] = b;
    nxt[cnt] = h[a];
    h[a] = cnt ++;
}

int n, m;
int re = 0;

void dfs(int u, int fa, int shu){
    if (shu > m) return; // 剪枝

    bool leaf = true;    // 先假设是叶子

    for (int i = h[u]; i != -1; i = nxt[i]){
        int v = t[i];
        if (v == fa) continue;

        leaf = false;    // 只要进了这个循环，说明有儿子，不是叶子

        int nshu;
        if (a[v] == 1){
            nshu = shu + 1;
        }else{
            nshu = 0;
        }

        dfs(v, u, nshu); // ★ 修复 1：传参改成 v, u
    }

    // ★ 修复 2：判断叶子的逻辑必须放在 for 循环的外面
    if (leaf) {
        re++;
    }
}

int main(){
    ios::sync_with_stdio(0);
    cin.tie(0);
    cnt = 0;
    memset(h, -1, sizeof(h));

    cin >> n >> m;

    // ★ 修复 3：下标从 1 开始
    for (int i = 1; i <= n; i++) cin >> a[i];

    for (int i = 1; i < n; i++){
        int u, v; // 建议不要用 a, b 做变量名，容易和上面的猫数组 a 混淆
        cin >> u >> v;
        add(u, v);
        add(v, u);
    }

    dfs(1, 0, a[1]);

    cout << re << endl;

    return 0;
}
    1689C - 被感染的树

    1689C - 被感染的树（Infected Tree）

    时间限制：3 秒

    内存限制：256 MB

    题目描述

    Byteland 是一片美丽的土地，以美丽的树而闻名。

    Misha 找到了一棵有 n 个顶点的二叉树，顶点编号为 1 到 n。二叉树是一个无环的连通无向图，包含 n 个顶点和 n − 1 条边。每个顶点的度数至多为 3，其中根是编号为 1 的顶点，根的度数至多为 2。

    不幸的是，根被感染了。

    接下来会发生 n 次如下过程：

    • Misha 要么选择一个未被感染且未被删除的顶点，将它以及所有与它相连的边删除；要么什么也不做。
    • 然后，感染会扩散到所有与已感染顶点通过一条边相连的顶点。所有已经感染的顶点仍然保持感染状态。

    由于 Misha 没有太多时间思考，请告诉他：最多能有多少个顶点免于感染？注意，被删除的顶点不算作被拯救。

    输入格式

    输入包含多组测试数据。

    第一行一个整数 t（1 ≤ t ≤ 5000），表示测试数据组数。

    接下来是每组测试数据。

    每组测试数据第一行一个整数 n（2 ≤ n ≤ 3 × 105），表示树的顶点数。

    接下来 n − 1 行，每行两个正整数 ui 和 vi（1 ≤ ui, vi ≤ n），表示它们之间有一条边。

    作为一个在数学上有天赋的男孩，他喜欢摆弄数字。这一次，他想在一棵有根树上摆弄数字。

    一棵树是一个没有环的连通图。一棵有根树有一个被称为根的特别顶点。节点 v 的父节点是从根到顶点 v 的路径上最后一个不同于 v 的顶点。顶点 v 的子节点是所有以 v 为父节点的节点。如果一个顶点没有子节点，那么它就是一个叶子节点。

    Serval 拥有的这棵有根树有 n 个节点，节点 1 是根。Serval 将在树的所有节点中写入一些数字。然而，有一些限制。除了叶子节点之外，每个节点中都写有一个运算 max 或 min，表示该节点中的数字应分别等于其所有儿子节点中数字的最大值或最小值。

    假设树中有 k 个叶子节点。Serval 想把整数 1, 2, …, k 放到这 k 个叶子节点上（每个数字必须恰好使用一次）。他喜欢大的数字，所以他希望最大化根节点中的数字。作为他最好的朋友，你能帮帮他吗？

    输入

    第一行包含一个整数 n（2 ≤ n ≤ 3·105），表示树的大小。

    第二行包含 n 个整数，其中第 i 个表示节点 i 中的运算。0 表示 min，1 表示 max。如果该节点是叶子节点，仍然会有一个 0 或 1 的数字，但你可以忽略它。

    第三行包含 n−1 个整数 f2, f3, …, fn（1 ≤ fi ≤ i−1），其中 fi 表示节点 i 的父节点。

    输出

    输出一个整数——根节点中可能的最大数字。

    1073. 树的中心

    1073. 树的中心

    题目描述

    给定一棵树，树中包含 n 个节点（编号为 1 到 n）和 n - 1 条无向边，每条边都有一个权值。

    请你在树中找到一个节点，使得该节点到树中其他节点的最远距离尽可能小。

    输入格式

    第一行包含一个整数 n。

    接下来的 n - 1 行，每行包含三个整数 ai、bi 和 ci，表示节点 ai 和节点 bi 之间存在一条权值为 ci 的边。

    输出格式

    输出一个整数，表示所求节点到树中其他节点的最远距离。

    数据范围

    • 1 ≤ n ≤ 10000
    • 1 ≤ ai, bi ≤ n

    输入样例：

    5
2 1 1
3 2 1
4 3 1
5 1 1

    输出样例：

    2
    树的中心.cpp
    #include<bits/stdc++.h>
using namespace std;

typedef long long ll;
#define endl "\n" 
const int N = 1e5 + 10;
const int M = 2 * N;
int h[N], nxt[M], t[M], w[M], cnt;
void add(int a, int b, int c){
    t[cnt] = b;
    w[cnt] = c;
    nxt[cnt] = h[a];
    h[a] = cnt ++;
}
ll d1[N], d2[N];
int son[N];
ll up[N]; 
void dfs(int u, int fa){
    for (int i = h[u]; i != -1; i = nxt[i]){        
        int v = t[i];
        if (v == fa)continue;
        dfs(v, u);  
        ll d = d1[v] + w[i];
        if (d > d1[u]){
            d2[u] = d1[u];
            d1[u] = d;
            son[u] = v;
        }else if(d > d2[u]){
            d2[u] = d;
        }
    }
}
void dfs_up(int u, int fa){
    for (int i = h[u]; i != -1; i = nxt[i]){
        int v = t[i];
        if (v == fa)continue;
        if (son[u] == v){
            up[v] = max(up[u], d2[u]) + w[i];
        }else {
            up[v] = max(up[u], d1[u]) + w[i];
        }
        dfs_up(v, u);
    }
}
int main(){
    ios::sync_with_stdio(0);
    cin.tie(0);
    cnt = 0;
    memset(h, -1, sizeof(h));
    int n;
    cin >> n;
    for (int i = 1; i < n; i++){
        int a, b, c;
        cin >> a >> b >> c;
        add(a, b, c);
        add(b, a, c);       
    }
    dfs(1, 0);
    up[1] = 0;
    dfs_up(1, 0);
    ll re = 1e18;
    for (int i = 1; i <= n; i++){
        re = min(re, max(d1[i], up[i]));
    }
    cout << re << endl;

    return 0;
}
